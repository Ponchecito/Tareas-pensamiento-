# LISTA
lista = [1, 2, 3, 4]

print("Lista original:", lista)

# Insertar dato
lista.append(5)
print("Lista con nuevo dato:", lista)


# TUPLA
tupla = (1, 2, 3)

print("Tupla:", tupla)

# Mostrar un dato
print("Primer dato de la tupla:", tupla[0])


# DICCIONARIO
diccionario = {
    "marca": "toyota",
    "modelo": "corolla",
    "anio": 2020
}

print("Diccionario original:")
print(diccionario)

# Insertar dato
diccionario["color"] = "rojo"

# Modificar dato
diccionario["anio"] = 2024

# Eliminar dato
del diccionario["modelo"]

print("Diccionario modificado:")
print(diccionario)
